# 定义


在uart 中  会注册设备节点的时候使用的是
register_chrdev_region 函数 ，注册有主设备号的设备节点


static struct file_operations hello_drv = {
	.owner	 = THIS_MODULE,
	.open    = hello_drv_open,
	.read    = hello_drv_read,
	.write   = hello_drv_write,
	.release = hello_drv_close,
};


分配一个区域
这里是 分配一个区域 包含主 设备号和 次设备号

dev_t devid;
int rc = alloc_chrdev_region(dev_t *dev, unsigned int firstminor, unsigned int count, const char *name)// 分配一个区域 包含主 设备号和 次设备号
int rc = alloc_chrdev_region(&devid, 0, 1, "hello"); // 分配一个区域 包含主 设备号和 次设备号
int major = MAJOR(devid); // 获取主设备号
int minor = MINOR(devid); // 获取次设备号
struct cdev cdev; // 定义 cdev 结构体
cdev_init(&cdev, hello_drv); // 初始化 cdev 结构体
cdev_add(&cdev, dev, 1); // 添加 cdev 结构体到内核
class_create(THIS_MODULE, name); // 创建 class
device_create(class, NULL, dev, NULL, name); // 创建设备节点


卸载
unregister_chrdev_region(dev_t dev, unsigned int count); // 卸载设备节点
cdev_del(&cdev); // 删除 cdev 结构体
class_destroy(class); // 删除 class
device_destroy(class, dev); // 删除设备节点
